# =============================================================
# Deploy Thai Legal RAG Chatbot ไปยัง Google Cloud Run
# =============================================================
# สถาปัตยกรรมบน cloud:
#   - Cloud Run (container จาก Dockerfile นี้) → เว็บแชท + LINE webhook + OpenClaw webhook
#   - Cloud LLM ผ่าน env: LLM_PROVIDER=openai (หรือ anthropic/google/openrouter)
#     * ถ้าอยากใช้ Ollama แบบ local บน cloud ให้ deploy Ollama เป็น service แยก
#       แล้วตั้ง OLLAMA_HOST ชี้ไปที่ URL ของ service นั้น
#   - Discord bot รันแยก process (websocket) — ดูหัวข้อท้ายไฟล์
#
# ข้อกำหนดเบื้องต้น:
#   gcloud components update
#   gcloud auth login
#   gcloud config set project <PROJECT_ID>

# -------------------------------------------------------------
# 1) เปิด APIs ที่จำเป็น (ครั้งเดียวต่อ project)
# -------------------------------------------------------------
gcloud services enable run.googleapis.com artifactregistry.googleapis.com cloudbuild.googleapis.com

# -------------------------------------------------------------
# 2) สร้าง Artifact Registry repo สำหรับเก็บ image (ครั้งเดียว)
# -------------------------------------------------------------
gcloud artifacts repositories create thai-law \
  --repository-format=docker \
  --location=asia-southeast1

# -------------------------------------------------------------
# 3) Build + Deploy ด้วย Cloud Build (ตรงไปตรงมาที่สุด)
#    แก้ PROJECT_ID / REGION / SERVICE ให้ตรงกับที่ใช้จริง
# -------------------------------------------------------------
PROJECT_ID=my-thai-law-project
REGION=asia-southeast1
SERVICE=thai-law-chatbot
IMAGE=${REGION}-docker.pkg.dev/${PROJECT_ID}/thai-law/${SERVICE}:v1

gcloud builds submit --tag ${IMAGE} .

gcloud run deploy ${SERVICE} \
  --image ${IMAGE} \
  --region ${REGION} \
  --port 8000 \
  --memory 2Gi \
  --cpu 2 \
  --min-instances 1 \
  --max-instances 5 \
  --timeout 300 \
  --concurrency 10 \
  --allow-unauthenticated \
  --set-env-vars "LLM_PROVIDER=openai,OPENAI_MODEL=gpt-4o-mini,OLLAMA_HOST=http://localhost:11434" \
  --set-secrets "OPENAI_API_KEY=openai-api-key:latest"

# -------------------------------------------------------------
# 4) สร้าง Secret ใน Secret Manager (ทำก่อนขั้น 3)
#    เก็บ API key ไว้ใน Secret Manager ไม่ใส่ env ตรงๆ
# -------------------------------------------------------------
echo -n "sk-..." | gcloud secrets create openai-api-key --data-file=-
gcloud secrets add-iam-policy-binding openai-api-key \
  --member="serviceAccount:$(gcloud projects describe ${PROJECT_ID} --format='value(projectNumber)')-compute@developer.gserviceaccount.com" \
  --role="roles/secretmanager.secretAccessor"

# ตัวอย่าง secret อื่นๆ (LINE / OpenClaw / Discord)
echo -n "<LINE_CHANNEL_ACCESS_TOKEN>" | gcloud secrets create line-channel-token --data-file=-
echo -n "<LINE_CHANNEL_SECRET>"     | gcloud secrets create line-channel-secret --data-file=-
# แล้วเพิ่มใน gcloud run deploy:
#   --set-secrets "OPENAI_API_KEY=openai-api-key:latest,LINE_CHANNEL_ACCESS_TOKEN=line-channel-token:latest,LINE_CHANNEL_SECRET=line-channel-secret:latest"

# -------------------------------------------------------------
# 5) Vector DB (chroma_db/)
# -------------------------------------------------------------
# chroma_db คือ filesystem ดังนั้นบน Cloud Run ต้องฝังใน image หรือใช้ชั้นเก็บข้อมูลแบบ network
# ทางเลือก:
#   A) COPY chroma_db/ ./chroma_db/ ใน Dockerfile แล้ว build+deploy ใหม่เมื่อ ingest เพิ่ม (ง่ายสุด)
#      เปิดคอมเมนต์บรรทัดนี้ใน Dockerfile แล้ว build ใหม่
#   B) ใช้ Cloud Storage เก็บ snapshot แล้ว download ตอน startup (แก้ src เพิ่ม)
#   C) ย้ายไปใช้ Chroma server mode / Cloud SQL สำหรับ pgvector (แก้ src/retriever.py)
#
# แนวทาง A คือ uncomment บรรทัด COPY chroma_db/ ใน Dockerfile ก่อน build:
#   COPY chroma_db/ ./chroma_db/

# -------------------------------------------------------------
# 6) อัปเดต env เพิ่มเติมภายหลัง (ไม่ต้อง build ใหม่)
# -------------------------------------------------------------
gcloud run services update ${SERVICE} \
  --region ${REGION} \
  --update-env-vars "LLM_PROVIDER=anthropic,ANTHROPIC_MODEL=claude-3-5-haiku-latest" \
  --update-secrets "ANTHROPIC_API_KEY=anthropic-api-key:latest"

# -------------------------------------------------------------
# 7) Custom domain + HTTPS (จำเป็นสำหรับ LINE webhook)
# -------------------------------------------------------------
gcloud beta run domain-mappings create --service ${SERVICE} --domain chat.example.com --region ${REGION}
# LINE webhook URL ที่ต้องใส่ใน LINE Developers Console:
#   https://chat.example.com/line/webhook

# -------------------------------------------------------------
# 8) Discord bot บน cloud (รันแยก ไม่ใช่ Cloud Run)
#    Discord ใช้ websocket outbound — ใช้ Cloud Run Jobs หรือ GCE/Compute Engine VM
#    แนะนำ: Compute Engine e2-small พอ:
# -------------------------------------------------------------
# gcloud compute instances create thai-law-discord \
#   --zone=asia-southeast1-b \
#   --machine-type=e2-small \
#   --image-family=debian-12 \
#   --image-project=debian-cloud \
#   --scopes=cloud-platform
#
# จากนั้น ssh เข้า VM:
#   git clone <repo> && cd <repo>
#   docker build -t thai-law-discord .
#   docker run -d --restart=always --env-file .env thai-law-discord \
#       python src/discord_bot.py
#
# (ทางเลือก) Cloud Run Jobs ก็รัน long-process ได้:
#   gcloud run jobs create thai-law-discord \
#     --image ${IMAGE} --region ${REGION} \
#     --command python --args src/discord_bot.py \
#     --set-cloudsql-instances "" 
#   gcloud beta run jobs executions deploy thai-law-discord --region ${REGION}

# -------------------------------------------------------------
# 9) Crawler ราชกิจจานุเบกษา ทุกสัปดาห์ (Cloud Scheduler → Cloud Run Job)
# -------------------------------------------------------------
gcloud run jobs create thai-law-crawler \
  --image ${IMAGE} --region ${REGION} \
  --command python --args src/crawl_ratchakitcha.py \
  --memory 1Gi --cpu 1 \
  --max-retries 1

gcloud scheduler jobs create http thai-law-crawler-weekly \
  --schedule "0 3 * * 1" \
  --uri "https://${REGION}-run.googleapis.com/apis/run.googleapis.com/v1/namespaces/${PROJECT_ID}/jobs/thai-law-crawler:run" \
  --http-method POST \
  --oauth-service-account-email $(gcloud iam service-accounts list --filter="compute@developer" --format="value(email)" | head -1)

# -------------------------------------------------------------
# 10) ทดสอบหลัง deploy
# -------------------------------------------------------------
SERVICE_URL=$(gcloud run services describe ${SERVICE} --region ${REGION} --format='value(status.url)')
curl -X POST ${SERVICE_URL}/chat \
  -H "Content-Type: application/json" \
  -d '{"session_id":"smoke-test","message":"ทดสอบการทำงาน"}'
curl ${SERVICE_URL}/line/health
