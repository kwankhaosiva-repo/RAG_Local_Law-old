# =============================================================
# Deploy ด้วย chroma_db จาก GCS (gs://chroma-db-law)
# Project: rag-law-509304
# =============================================================
# ต่างจาก deploy ปกติตรงที่ container จะโหลด chroma_db จาก GCS เป็น
# BACKGROUND หลัง bind PORT แล้ว (แทนการ COPY เข้า image ซึ่งจะใหญ่ 3.4GB+)
#
# ขั้นตอน:
# 1) upload chroma_db ขึ้น GCS (ทำแล้ว):
#      gcloud storage cp -r ./chroma_db gs://chroma-db-law/
#
# 2) เปิด API ที่จำเป็น (ครั้งเดียว):
#      gcloud services enable run.googleapis.com storage.googleapis.com \
#        firestore.googleapis.com bigquery.googleapis.com --project rag-law-509304
#
# 3) ให้ Cloud Run service account อ่าน bucket ได้:
#      PROJECT_ID=rag-law-509304
#      PROJECT_NUMBER=$(gcloud projects describe ${PROJECT_ID} --format='value(projectNumber)')
#      gcloud storage buckets add-iam-policy-binding gs://chroma-db-law \
#        --member="serviceAccount:${PROJECT_NUMBER}-compute@developer.gserviceaccount.com" \
#        --role="roles/storage.objectViewer"
#
# 4) สร้าง Firestore database (ทำแล้ว: rag-law-db) และ BigQuery dataset (ครั้งเดียว):
#      bq mk --dataset --location=asia-southeast1 rag-law-509304:rag_law_query
#      bq mk --table rag-law-509304:rag_law_query.queries \
#        session_id:STRING,question:STRING,answer:STRING,sources_json:STRING,created_at:TIMESTAMP
#
# 5) deploy — สังเกต --port 8080 (ต้องตรงกับ PORT ใน container ห้ามใช้ 8000):
#      gcloud run deploy rag-law \
#        --image ${IMAGE} \
#        --project rag-law-509304 \
#        --region ${REGION} \
#        --port 8080 \
#        --memory 2Gi --cpu 2 \   # *** ห้ามใช้ default 512Mi — embedding model ใช้ RAM ~1GB → OOM kill
#                                 #    (อาการ: POST /chat → 503 + "Memory limit exceeded" ใน log)
#                                 #    โหมด pinecone ใช้ 2Gi พอ / โหมด chroma ต้อง 4Gi
#        --min-instances 1 --max-instances 5 \
#        --timeout 300 --concurrency 10 \
#        --allow-unauthenticated \
#        --set-env-vars "LLM_PROVIDER=gateway,GATEWAY_BASE_URL=https://unorouter.com/v1,GATEWAY_MODEL=gpt-4o-mini,GCS_CHROMA_BUCKET=gs://chroma-db-law,GCS_CHROMA_PREFIX=chroma_db,FIRESTORE_DB=rag-law-db" \
#        --set-secrets "ENV_LAW=env_law:latest"
#
# *** LLM setup (Gemini ปิดไว้ก่อน — free key ต้องผูก billing) ***
# ใช้ UNOROUTER เป็น primary แล้วมี failover chain ต่ออัตโนมัติ:
#   LLM_PROVIDER=unorouter  (หรือ LLM_PROVIDER=gateway + GATEWAY_* ก็ได้)
#   default chain: unorouter → groq → mistral → ollama
#   (provider ที่ไม่มี key ถูกข้ามอัตโนมัติ — ใส่ GROQ_API/MISTRAL_API เพิ่มใน env_law
#    ได้เลย ไม่ต้องแก้ code)
# แบบ local Ollama ผ่าน ngrok: ดู docs/ngrok_local_llm.txt
#
# เพิ่ม GATEWAY_API_KEY / GROQ_API / MISTRAL_API ลงใน secret env_law ได้เลย เช่น:
#   GATEWAY_BASE_URL=https://unorouter.com/v1
#   GATEWAY_API_KEY=...
#   GATEWAY_MODEL=gpt-4o-mini
#   GROQ_API=gsk_...
#   MISTRAL_API=...
#   LLM_FAILOVER_CHAIN=groq,mistral
#
# secret "env_law" เก็บ env ทั้งหมดเป็นก้อนเดียว (รูปแบบ .env) เช่น:
#   OPENAI_API_KEY=sk-...
#   LINE_CHANNEL_ACCESS_TOKEN=...
#   LINE_CHANNEL_SECRET=...
# โค้ด (src/config.py บนสุด) จะ parse ให้เองตอน import
# หมายเหตุ: ENV_LAW (ชื่อ env var ฝั่ง container) ตั้งเป็นตัวพิมพ์ใหญ่
#   secret name ที่ Secret Manager = env_law (ตามที่สร้างไว้)
#
# 6) ทดสอบหลัง deploy:
#      SERVICE_URL=$(gcloud run services describe rag-law --region ${REGION} \
#        --project rag-law-509304 --format='value(status.url)')
#      curl ${SERVICE_URL}/health
#      curl -X POST ${SERVICE_URL}/chat -H "Content-Type: application/json" \
#        -d '{"session_id":"smoke-test","message":"ทดสอบการทำงาน"}'
#
# หมายเหตุสำคัญ:
# - chroma_db ถูกโหลดจาก GCS แบบ BACKGROUND หลัง container bind PORT แล้ว
#   (Cloud Run health check ผ่านทันที ไม่ fail "container failed to start")
# - ตรวจสถานะ sync ได้ที่ GET /health → {"db_ready": true/false}
# - คำถามแรกๆ ที่มาก่อน sync เสร็จ จะรอจน DB พร้อม (สูงสุด 5 นาที)
# - rsync ครั้งแรกโหลดเต็ม ~3.4GB (region เดียวกับ bucket = เร็ว), ครั้งถัดๆ ไปเฉพาะไฟล์ที่เปลี่ยน
# - memory แนะนำ 4Gi เพราะต้องโหลด index ทั้งก้อนมาทำงาน
#
# --- ทางเลือก: Pinecone (ตัด cold start ที่มาจากการ sync chroma_db ออก) ---
# 1) migrate ข้อมูล (รัน local ครั้งเดียว):
#      export PINECONE_KEY=pcsk_...   # (ใส่ใน secret env_law แล้ว)
#      pip install pinecone langchain-pinecone
#      python scripts/migrate_chroma_to_pinecone.py --dry-run   # ดูจำนวนก่อน
#      python scripts/migrate_chroma_to_pinecone.py             # อัปจริง
# 2) deploy ด้วย env เพิ่ม:
#      VECTOR_STORE=pinecone
#    → container จะข้ามการ rsync chroma_db จาก GCS ทันที startup ไวขึ้น
#    → ไม่ต้องให้ SA มีสิทธิ์อ่าน bucket, ไม่ต้องพึ่ง gcloud CLI ใน image
# 3) หมายเหตุ: BM25 hybrid search ยังทำงานเหมือนเดิม (rerank ใน memory)
#
# --- Firestore (rag-law-db) ---
# - ประวัติแชททุกเทิร์นเก็บที่ collection "chat_sessions" → doc <session_id> → subcollection "turns"
# - ดูใน console: Firestore Studio → database rag-law-db
# - กด /reset (เว็บหรือพิมพ์ /reset ใน LINE) จะลบ turns ของ session นั้นด้วย
# - ถ้า Firestore ใช้ไม่ได้ (API ปิด/สิทธิ์ไม่พอ) ระบบยังตอบแชทได้ปกติ แค่ไม่เก็บประวัติ
#
# --- BigQuery (rag_law_query.queries) ---
# - ทุกคำถาม/คำตอบ insert ลงตาราง queries ไว้วิเคราะห์ เช่น คำถามยอดนิยม:
#     SELECT question, COUNT(*) c FROM `rag-law-509304.rag_law_query.queries`
#     GROUP BY question ORDER BY c DESC LIMIT 20
# - ถ้า BigQuery ใช้ไม่ได้ ก็ข้ามไปเฉยๆ ไม่กระทบผู้ใช้
#
# LINE webhook หมายเหตุ:
# - webhook ตอบ 200 ทันที แล้วประมวลผล LLM ใน background (LINE timeout = 2 วิ)
# - ใช้ Reply API (ฟรี ไม่นับ quota) ถ้า token หมดอายุจะ fallback ไป Push API (นับ quota)
# - ใน LINE Developers Console: Webhook URL = https://<cloud-run-url>/line/webhook
#   ปิด auto-reply/greeting ของ Official Account และอย่าเปิด "Use grouped responses"
